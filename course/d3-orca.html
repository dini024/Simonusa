D3.2 Contoh kerja

1 · Optimasi + frekuensi
! B3LYP D3BJ def2-SVP Opt Freq TightSCF
%pal nprocs 8 end
%maxcore 2000
* xyzfile 0 1 mol.xyz
2 · TD-DFT (struktur sudah optimasi)
! CAM-B3LYP def2-TZVP CPCM(toluene) RIJCOSX def2/J TightSCF
%pal nprocs 8 end
%maxcore 2000
%tddft nroots 20
 tda false
end
* xyzfile 0 1 opt.xyz

Jalankan: orca job.inp > job.out. ORCA paralel harus dipanggil dengan path lengkap: $(which orca) job.inp > job.out.

Skrip SLURM (sesuaikan modul/partisi cluster Anda)
#!/bin/bash
#SBATCH -J orca_job
#SBATCH -N 1
#SBATCH -n 8
#SBATCH -o slurm-%j.log
module load orca            # nama modul tergantung cluster
cd $SLURM_SUBMIT_DIR
$(which orca) job.inp > job.out
